\subsection*{Temperature $T(m,t)$}
In the shocked region, the temperature profile $T(m,t)$ is derived algebraically from the equation of state (EOS) along with the well-behaved fits for pressure $p(m,t)$ and density $\rho(m,t)$:
\begin{enumerate}
    \item \textbf{Theoretical relation}:
    $$T(m,t) = \left( \frac{p(m,t) \, v(m,t)^{1-\mu}}{r \, f} \right)^{\frac{1}{\beta}} = \left( \frac{p(m,t) \, \rho(m,t)^{\mu-1}}{r \, f} \right)^{\frac{1}{\beta}}$$
    \item \textbf{CGS Parametrized form}:
    $$T(m,t) \approx \left( \frac{P(y_s) \, p_0 \, t^\tau \left( \rho_0 \, \rho_s \, y_s^{d_\rho} \right)^{\mu-1}}{r \, f_{\text{Kelvin}}} \right)^{\frac{1}{\beta}}$$
    \item \textbf{Algebraic simplification of bracket}:
    We pull the power $y_s^{(\mu-1)}$ out of the bracket raised to $1/\beta = 0.625$, grouping all constants:
    $$T(y_s, t) = T_{\text{scale}} \cdot y_s^{-0.7125 \, d_\rho} \left[ 1.0 + (P_s - 1.0) \, y_s^{d_p} \right]^{0.625} \, t^{-\frac{215}{768}}$$
    \item \textbf{Substitution of scaling constants and units}:
    Using evaluated prefactors ($T_{\text{scale\_hev}} \approx 0.01240 \text{ HeV}$):
    $$T(m,t) \approx 0.01240 \cdot y_s^{-0.46269} \left[ 1.0 + 0.49535 \, y_s^{1.13315} \right]^{0.625} \, \left(\frac{t}{\text{ns}}\right)^{-\frac{215}{768}} \quad \text{HeV}$$
    $$\text{Or in Kelvin: } T(m,t) \approx 14393.59038 \cdot y_s^{-0.46269} \left[ 1.0 + 0.49535 \, y_s^{1.13315} \right]^{0.625} \, \left(\frac{t}{\text{ns}}\right)^{-\frac{215}{768}} \quad \text{Kelvin}$$
    \item \textbf{Fully Substituted CGS function}:

    $$T(m,t) \approx 0.00161 \, \left[ \left(\frac{m}{\text{g/cm}^2}\right)^{-0.74030} \left(\frac{t}{\text{ns}}\right)^{0.57450} + 73.73330 \, \left(\frac{m}{\text{g/cm}^2}\right)^{0.39285} \left(\frac{t}{\text{ns}}\right)^{-0.30487} \right]^{0.625} \, \left(\frac{t}{\text{ns}}\right)^{-\frac{215}{768}} \quad \text{HeV}$$

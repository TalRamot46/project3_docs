\subsection{Wave Coordinate Normalization Variables}
The ablated mass coordinate $m_f(t)$ and shocked mass coordinate $m_s(t)$ grow as exact power laws in time:
\begin{gather}
    m_f(t) = \xi_f A^{-a} B^{-b} t^{-c} \approx 0.00102 \left(\frac{t}{\text{ns}}\right)^{\frac{33}{64}} \quad \text{g/cm}^2 \\
    m_s(t) = m_{s0} \left(\frac{t}{\text{ns}}\right)^{\frac{149}{192}} \approx 0.01209 \left(\frac{t}{\text{ns}}\right)^{\frac{149}{192}} \quad \text{g/cm}^2
\end{gather}
where $\xi_s$ is the shock similarity front coordinate:
\begin{equation}
    \xi_s = 1.67133
\end{equation}
So we can define the normalized mass coordinates $y = m/m_f(t)$ and $y_s = m/m_s(t)$ directly as:
\begin{gather}
    y = \frac{m}{m_f(t)} \approx 981.45247 \left(\frac{m}{\text{g/cm}^2}\right) \left(\frac{t}{\text{ns}}\right)^{-\frac{33}{64}} \\
    y_s = \frac{m}{m_s(t)} \approx 82.68924 \left(\frac{m}{\text{g/cm}^2}\right) \left(\frac{t}{\text{ns}}\right)^{-\frac{149}{192}}
\end{gather}

\subsection{Density $\rho(m,t)$}
In the shock compressed region under the \texttt{FIT\_RHO\_AROUND\_PISTON} option, the density profile is piecewise. Near the piston ($y_s < 0.1$), the density is fitted directly to capture the high-compression boundary behavior. Near the shock front ($y_s \ge 0.1$), the density is derived algebraically from the equation of state (EOS) using the fitted temperature and pressure profiles to ensure thermodynamic consistency.
\begin{enumerate}
    \item \textbf{Dimensionless Fit Function}:
    $$\rho(y_s) \approx \begin{cases} 
        \rho_s + (\rho_0 - \rho_s) (1 - y_s^{0.45561})^{0.69201} & \text{for } y_s < 0.1 \\
        \left( \frac{P(y_s)}{r \, f_{\text{dim}} \, T(y_s)^\beta} \right)^{\frac{1}{1-\mu}} & \text{for } y_s \ge 0.1 
    \end{cases}$$
    
    \item \textbf{Theoretical Scaling}:
    $$\rho(m,t) = \frac{1}{v_0} \, \rho(y_s) = \rho_0 \, \rho(y_s)$$
    
    \item \textbf{Substitution of scaling constants and units}:
    Using physical density $\rho_0 = 19.32 \text{ g/cm}^3$:
    $$\rho(m,t) \approx 19.32 \, \rho(y_s) \quad \text{g/cm}^3$$
    
    \item \textbf{Fully Substituted CGS function}:
    Substituting $y_s \approx 82.68924 \left(\frac{m}{\text{g/cm}^2}\right) \left(\frac{t}{\text{ns}}\right)^{-\frac{149}{192}}$:
    $$\rho(m,t) \approx \begin{cases}
        19.32000 \left[ 9.00000 + (0.00012 - 9.00000) (1 - y_s^{0.45561})^{0.69201} \right] & \text{for } y_s < 0.1 \\
        \left( \frac{p(m,t)}{r \, f_{\text{Kelvin}} \, T(m,t)^\beta} \right)^{\frac{1}{1-\mu}} & \text{for } y_s \ge 0.1 
    \end{cases} \quad \text{g/cm}^3$$
\end{enumerate}

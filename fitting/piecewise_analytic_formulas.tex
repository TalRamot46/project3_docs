\subsection{Wave Coordinate Normalization Variables}
The ablated mass coordinate $m_f(t)$ and shocked mass coordinate $m_s(t)$ grow as exact power laws in time:
\begin{gather}
    m_f(t) = \xi_f A^{-a} B^{-b} t^{-c} \approx 0.00102 \left(\frac{t}{\text{ns}}\right)^{\frac{33}{64}} \quad \text{g/cm}^2 \\
    m_s(t) = m_{s0} \left(\frac{t}{\text{ns}}\right)^{\frac{149}{192}} \approx 0.01209 \left(\frac{t}{\text{ns}}\right)^{\frac{149}{192}} \quad \text{g/cm}^2
\end{gather}
where $\xi_s$ is the shock similarity front coordinate:
\begin{equation}
    \xi_s = 1.67133
\end{equation}
So we can define the normalized mass coordinates $y = m/m_f(t)$ and $y_s = m/m_s(t)$ directly as:
\begin{gather}
    y = \frac{m}{m_f(t)} \approx 981.45247 \left(\frac{m}{\text{g/cm}^2}\right) \left(\frac{t}{\text{ns}}\right)^{-\frac{33}{64}} \\
    y_s = \frac{m}{m_s(t)} \approx 82.68924 \left(\frac{m}{\text{g/cm}^2}\right) \left(\frac{t}{\text{ns}}\right)^{-\frac{149}{192}}
\end{gather}

\subsection{Density $\rho(m,t)$}
In the shock compressed region under the \texttt{FIT\_RHO\_AROUND\_PISTON} option, the density profile is piecewise. Near the piston ($y_s < 0.1$), the density is fitted directly to capture the high-compression boundary behavior. Near the shock front ($y_s \ge 0.1$), the density is derived algebraically from the equation of state (EOS) using the fitted temperature and pressure profiles to ensure thermodynamic consistency.
\begin{enumerate}
    \item \textbf{Dimensionless Fit Function}:
    $$\rho(y_s) \approx \begin{cases} 
        \rho_s + (\rho_0 - \rho_s) (1 - y_s^{0.45561})^{0.69201} & \text{for } y_s < 0.1 \\
        \left( \frac{P(y_s)}{r \, f_{\text{dim}} \, T(y_s)^\beta} \right)^{\frac{1}{1-\mu}} & \text{for } y_s \ge 0.1 
    \end{cases}$$
    
    \item \textbf{Theoretical Scaling}:
    $$\rho(m,t) = \frac{1}{v_0} \, \rho(y_s) = \rho_0 \, \rho(y_s)$$
    
    \item \textbf{Substitution of scaling constants and units}:
    Using physical density $\rho_0 = 19.32 \text{ g/cm}^3$:
    $$\rho(m,t) \approx 19.32 \, \rho(y_s) \quad \text{g/cm}^3$$
    
    \item \textbf{Fully Substituted CGS function}:
    Substituting $y_s \approx 82.68924 \left(\frac{m}{\text{g/cm}^2}\right) \left(\frac{t}{\text{ns}}\right)^{-\frac{149}{192}}$:
    $$\rho(m,t) \approx \begin{cases}
        19.32000 \left[ 9.00000 + (0.00012 - 9.00000) (1 - y_s^{0.45561})^{0.69201} \right] & \text{for } y_s < 0.1 \\
        \left( \frac{p(m,t)}{r \, f_{\text{Kelvin}} \, T(m,t)^\beta} \right)^{\frac{1}{1-\mu}} & \text{for } y_s \ge 0.1 
    \end{cases} \quad \text{g/cm}^3$$
\end{enumerate}

\section{Eulerian Coordinate Transformations}
In the Lagrangian coordinate system, physical variables are expressed in terms of the mass coordinate $m$ and time $t$. To transition to Eulerian coordinates, we define the absolute spatial position $x(m,t)$ by integrating the specific volume $v(m,t)$ relative to a reference front location:
\begin{equation}
    x(m,t) = x_{\text{ref}}(t) + \int_{m_{\text{ref}}}^{m} v(m',t) \, dm'
\end{equation}

\subsection{Wave Front Trajectories and Position Scales}
The time-dependent trajectories of the boundaries and shock interfaces are expressed using fully substituted physical constants and scales evaluated at $t = 1\text{ ns}$.

\begin{enumerate}
    \item \textbf{Ablation Front / Piston Interface position} $x_{af}(t)$:
    The ablation front lies at the boundary of the subsonic region, which maps to $y_s = y_{mf} \approx 0.08425$ in the shock coordinates. Its position scale is determined by the shock integration factor:
    \begin{equation}
        x_{af}(t) = \text{pos\_scale}_s(t) \left[ (1-\omega)\xi_{mf} V(y_{mf}) + \frac{2-\omega}{\tau_s+2} U(y_{mf}) \right]
    \end{equation}
    Substituting physical values yields the CGS trajectory:
    \begin{equation}
        x_{af}(t) \approx 5.12397 \times 10^{-4} \left(\frac{t}{\text{ns}}\right)^{\frac{149}{192}} \quad \text{cm}
    \end{equation}

    \item \textbf{Ablation Boundary position} $x_b(t)$:
    The ablation boundary corresponds to the zero-mass coordinate $m=0$. Its position is obtained by integrating specific volume from the ablation front to $m=0$:
    \begin{equation}
        x_b(t) = x_{af}(t) + C_{\text{pos}}(t) \, U_0
    \end{equation}
    where $C_{\text{pos}}(t) = \frac{A^{a_u} B^{b_u} t^{c_u+1}}{c_u+1}$. Substituting the physical values (with $U_0 \approx -9.08174$) yields:
    \begin{equation}
        x_b(t) \approx 5.12397 \times 10^{-4} \left(\frac{t}{\text{ns}}\right)^{\frac{149}{192}} - 3.01147 \times 10^{-2} \left(\frac{t}{\text{ns}}\right)^{\frac{199}{192}} \quad \text{cm}
    \end{equation}

    \item \textbf{Shock Piston position} $x_p(t)$:
    The piston driving the shock corresponds to the inner boundary of the shock-compressed region ($m = 0$ in shock mass coordinates, or $y_s = 0$). Its position trajectory scales as:
    \begin{equation}
        x_p(t) = \text{pos\_scale}_s(t) \left( \frac{2-\omega}{\tau_s+2} \right) U(0)
    \end{equation}
    Substituting the initial shock velocity factor $U_0 \approx + 0.96550$ yields:
    \begin{equation}
        x_p(t) \approx 4.65959 \times 10^{-4} \left(\frac{t}{\text{ns}}\right)^{\frac{149}{192}} \quad \text{cm}
    \end{equation}

    \item \textbf{Shock Front position} $x_s(t)$:
    The outer edge of the shock wave ($m = m_s(t)$, or $y_s = 1$) moves according to:
    \begin{equation}
        x_s(t) = \text{pos\_scale}_s(t) \left[ (1-\omega)\xi_s V_s + \frac{2-\omega}{\tau_s+2} U_s \right]
    \end{equation}
    Substituting the shock coordinate $\xi_s = 1.67133$ and the fitted boundary values yields:
    \begin{equation}
        x_s(t) \approx 6.25956 \times 10^{-4} \left(\frac{t}{\text{ns}}\right)^{\frac{149}{192}} \quad \text{cm}
    \end{equation}
\end{enumerate}

\section{Integrated Energies}
Integrated energies per unit area are computed piecewise across the subsonic ablation region and the shock compressed region.

\subsection{Subsonic Ablation Region ($m < m_f(t)$)}
\begin{enumerate}
    \item \textbf{Kinetic Energy} $E_k(t)$:
    \begin{equation}
        E_k(t) = A^{2a_u - a} B^{2b_u - b} t^{2c_u - c} \int_0^{\xi_f} \frac{U^2(\xi)}{2} d\xi
    \end{equation}
    Substituting numerical values yields:
    \begin{equation}
        E_k(t) \approx 2.22491 \times 10^{10} \left(\frac{t}{\text{ns}}\right)^{\frac{113}{192}} \quad \text{erg/cm}^2
    \end{equation}

    \item \textbf{Internal Energy} $E_{in}(t)$:
    \begin{equation}
        E_{in}(t) = A^{2a_u - a} B^{2b_u - b} t^{2c_u - c} \int_0^{\xi_f} \frac{P(\xi)V(\xi)}{r} d\xi
    \end{equation}
    Substituting numerical values yields:
    \begin{equation}
        E_{in}(t) \approx 3.61070 \times 10^{10} \left(\frac{t}{\text{ns}}\right)^{\frac{113}{192}} \quad \text{erg/cm}^2
    \end{equation}

    \item \textbf{Total Energy} $E(t)$:
    \begin{equation}
        E(t) = A^{2a_u - a} B^{2b_u - b} t^{2c_u - c} \frac{S_0}{2c_u - c}
    \end{equation}
    Substituting numerical values yields:
    \begin{equation}
        E(t) \approx 5.83561 \times 10^{10} \left(\frac{t}{\text{ns}}\right)^{\frac{113}{192}} \quad \text{erg/cm}^2
    \end{equation}
\end{enumerate}

\begin{enumerate}
